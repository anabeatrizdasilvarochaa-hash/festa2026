const SHEET_NAME = "Confirmacoes"; function setup() { const props = PropertiesService.getScriptProperties(); let id = props.getProperty("SPREADSHEET_ID"); let ss; if (id) { ss = SpreadsheetApp.openById(id); } else { ss = SpreadsheetApp.create("Festa de Final de Ano 2026 - Confirmacoes"); props.setProperty("SPREADSHEET_ID", ss.getId()); } let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) sheet = ss.insertSheet(SHEET_NAME); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Nome do participante", "Time", "Cargo", "Valor da cota", "Data e horário da confirmação" ]); sheet.setFrozenRows(1); sheet.getRange(1,1,1,5).setFontWeight("bold"); sheet.getRange("D:D").setNumberFormat('"R$" #,##0.00'); sheet.getRange("E:E").setNumberFormat("dd/mm/yyyy hh:mm:ss"); } return ss.getUrl(); } function doPost(e) { try { const body = e && e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {}; const name = String(body.name || "").trim().slice(0,120); const team = String(body.team || ""); const role = String(body.role || ""); const allowedTeams = [ "Cobrança", "Controladoria", "Fiscal", "Jurídico", "MIS", "Op. Financeiras", "Planej. Estratégico", "Planej. Financeiro", "RI & M&A", "Suprimentos", "Visagio" ]; const amounts = { "Assistente/Estagiário/Jovem Aprendiz": 0, "Analista": 60, "Especialista": 110, "Coordenador": 180 }; if ( name.length < 2 || !allowedTeams.includes(team) || !Object.prototype.hasOwnProperty.call(amounts, role) ) { return jsonResponse({ok:false, message:"Dados inválidos."}); } const id = PropertiesService.getScriptProperties() .getProperty("SPREADSHEET_ID"); if (!id) { return jsonResponse({ok:false, message:"Execute setup() primeiro."}); } const sheet = SpreadsheetApp.openById(id).getSheetByName(SHEET_NAME); sheet.appendRow([ name, team, role, amounts[role], new Date() ]); return jsonResponse({ok:true}); } catch (err) { return jsonResponse({ ok:false, message:"Não foi possível registrar." }); } } function doGet() { return jsonResponse({ ok:true, service:"Festa de Final de Ano 2026 - registro de presença" }); } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
